One Standard, Five Native Implementations

KVDock executes with identical zero-dependency semantics across edge runtimes, serverless handlers, and distributed microservices. 

// npm install kvdock@0.3.2
import { KVDock, TieredDock } from "kvdock";
import { MemoryAdapter } from "kvdock/adapters/memory";
import { RedisAdapter } from "kvdock/adapters/redis";

// Pair ultra-fast in-memory L1 with persistent L2
const dock = new TieredDock({
  l1: new MemoryAdapter({ maxSize: 10_000 }),
  l2: new RedisAdapter({ url: "redis://localhost:6379" }),
  l1TTL: 15_000 // Micro-cache hot keys for 15s
});

// Single-Flight Stampede Protection: 1,000 requests -> 1 DB fetch
const product = await dock.getOrSet("product:catalog:hot", async () => {
  return await db.query("SELECT * FROM catalog WHERE hot = true");
}, { ttl: 300_000 });
